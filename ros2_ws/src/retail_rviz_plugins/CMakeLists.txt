cmake_minimum_required(VERSION 3.16)
project(retail_rviz_plugins)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(retail_interfaces REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
set(CMAKE_AUTOMOC ON)
option(BUILD_TESTING "Build isolated RViz controls smoke test" ON)
add_library(retail_controls SHARED src/controls_panel.cpp)
target_compile_features(retail_controls PUBLIC cxx_std_17)
ament_target_dependencies(retail_controls rclcpp rviz_common pluginlib std_msgs std_srvs nav_msgs retail_interfaces)
target_link_libraries(retail_controls Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS retail_controls LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  add_executable(controls_smoke test/controls_smoke.cpp)
  target_compile_features(controls_smoke PUBLIC cxx_std_17)
  ament_target_dependencies(controls_smoke rclcpp rviz_common pluginlib std_msgs std_srvs retail_interfaces)
  target_link_libraries(controls_smoke Qt5::Widgets)
  install(TARGETS controls_smoke DESTINATION lib/${PROJECT_NAME})
endif()
ament_package()
